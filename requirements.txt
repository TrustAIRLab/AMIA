# ─── Production-memory frameworks (production_memory/ only) ─────────────────────
# Exact pins below match the paper environment (the May-2026 framework run).
# mem0ai is pinned to 2.0.2 (the version current at the time of the experiments);
# its fact-extractor wording differs in later 2.0.x, which shifts the Mem0 ReAct
# numbers, so keep this pin to reproduce the paper's Mem0 results. sentence-
# transformers is the one that matters for the other backends (A-MEM / MemoryOS
# embeddings); the rest are low-impact (faiss IndexFlatIP is exact brute-force,
# litellm is a transport layer, BM25 is deterministic).
mem0ai==2.0.2                    # Mem0
qdrant-client==1.18.0            # Mem0 vector store
sentence-transformers==2.7.0     # A-MEM + MemoryOS embeddings (all-MiniLM-L6-v2) — paper env
nltk==3.10.0                     # A-MEM tokenization
scikit-learn==1.9.0              # A-MEM cosine similarity
faiss-cpu==1.15.1                # MemoryOS mid-term IndexFlatIP search (exact; version-robust)
chromadb==1.5.9                  # A-MEM vector store
litellm==1.83.0                  # A-MEM internal LLM client
rank-bm25==0.2.2                 # A-MEM BM25 retrieval
